import axios from "axios";
import Redis from "ioredis";

const redis = new Redis();

export async function publishBinanceMovers() {
  const { data } = await axios.get(
    "https://api.binance.com/api/v3/ticker/24hr"
  );

  const sorted = data
    .filter(t => t.symbol.endsWith("USDT"))
    .sort((a, b) => Math.abs(b.priceChangePercent) - Math.abs(a.priceChangePercent));

  const gainers = sorted
    .filter(t => Number(t.priceChangePercent) > 0)
    .slice(0, 10)
    .map(t => t.symbol.replace("USDT", ""));

  const losers = sorted
    .filter(t => Number(t.priceChangePercent) < 0)
    .slice(0, 10)
    .map(t => t.symbol.replace("USDT", ""));

  await redis.publish(
    "movers:binance",
    JSON.stringify({
      timestamp: Date.now(),
      gainers,
      losers,
      source: "binance"
    })
  );
}

6️⃣ SCANNER — SUBSCRIBE TO MOVERS
📁 scanner/src/movers.ts
import Redis from "ioredis";

const redis = new Redis();

const moversByExchange = new Map<string, Set<string>>();

export function initMoversSubscription() {
  redis.psubscribe("movers:*");

  redis.on("pmessage", (_, channel, message) => {
    const exchange = channel.split(":")[1];
    const payload = JSON.parse(message);

    const assets = new Set([
      ...payload.gainers,
      ...payload.losers
    ]);

    moversByExchange.set(exchange, assets);
  });
}

export function getPriorityAssets(exchange: string): string[] {
  return Array.from(moversByExchange.get(exchange) || []);
}

7️⃣ DYNAMIC PATH GENERATION (CRITICAL FIX)

import { getPriorityAssets } from "./movers";

const BASE_QUOTES = ["USDT", "BTC", "ETH", "BNB"];

export function generateDynamicPaths(
  exchange: string,
  availableSymbols: string[]
) {
  const dynamicAssets = new Set([
    ...BASE_QUOTES,
    ...getPriorityAssets(exchange)
  ]);

  const graph: Record<string, Set<string>> = {};

  for (const symbol of availableSymbols) {
    for (const quote of BASE_QUOTES) {
      if (symbol.endsWith(quote)) {
        const base = symbol.replace(quote, "");
        graph[base] ??= new Set();
        graph[quote] ??= new Set();
        graph[base].add(quote);
        graph[quote].add(base);
      }
    }
  }

  const paths: string[][] = [];

  for (const a of dynamicAssets) {
    for (const b of graph[a] || []) {
      for (const c of graph[b] || []) {
        if (c !== a && graph[c]?.has(a)) {
          paths.push([a, b, c]);
        }
      }
    }
  }

  return paths;
}
